cmake_minimum_required(VERSION 3.20)
project(ltm_knn_measurement LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(LTM_BUILD_TESTS OFF CACHE BOOL "Build core tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
add_executable(ltm_knn_bench knn_bench.cc)
target_link_libraries(ltm_knn_bench PRIVATE ltm_index)
foreach(prop IN ITEMS COMPILE_OPTIONS COMPILE_DEFINITIONS LINK_OPTIONS CXX_COMPILER_LAUNCHER)
  get_target_property(value ltm_core ${prop})
  if(value)
    set_property(TARGET ltm_knn_bench PROPERTY ${prop} "${value}")
  endif()
endforeach()
set_target_properties(ltm_knn_bench PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
